  134. function wrt_cfg_str($c, $file) {
  135.   $err = 0;
  136.   $f = fopen($file, 'wb');
  137.   if($f !== false) {
  138.     $w = fwrite($f, $c);
  139.     if ($w === false) {$err = 1;} // could not write
  140.     else {$err = 0;} // write ok
  141.     fclose($f);
  142.   }
  143.   else {$err = 2;} // could not open file for writing
  144.   return $err;

  158. /**
  159. * Split sql file into separate queries
  160. *
  161. * inspired by phpbb code
  162. *
  163. * return mixed stuff
  164. */
  165. function split_sql($file, $delimiter) {
  166.   $sql = file_get_contents($file);
  167.   // trim
  168.   $sql = trim($sql);
  169.     // remove # comments
  170.   $sql = preg_replace('/\n{2,}/', "\n", preg_replace('/^#.*$/m', "\n", $sql));
  171.   // remove -- comments
  172.     $sql = preg_replace('/\n{2,}/', "\n", preg_replace('/^--.*$/m', "\n", $sql));
  173.   // remove \r
  174.   $sql = eregi_replace("\r", "", $sql);
  175.   // split
  176.   $sql = preg_split('/' . preg_quote($delimiter, '/') . '$/m', $sql);
  177.   // trim pieces
  178.   $sql = array_map('trim', $sql);
  179.   // delete last one if empty
  180.   $last_one = end($sql);
  181.   if(empty($last_one)) {unset($sql[key($sql)]);}
  182.   // return
  183.   return $sql;
  184. }
