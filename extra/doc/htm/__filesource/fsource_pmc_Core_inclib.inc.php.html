  160. /**
  161. * Create Search Engine friendly URLs
  162. *
  163. * before: index.php?id=2&page=3
  164. * after: index.php/id/2/page/3
  165. * inspiration from code snippets in the php manual
  166. */
  167. function pmc_seo_urls() {
  168.   // we will not be taking orders from normal style requests!
  169.   // ignore ?id=2&page=3
  170.   // we didn't do our stuff yet, so if GET exists it means it has values from normal style requests
  171.   if(!empty($_GET)) {$_GET = array();}
  172.   // if the request is in the form id/2/page/3 put them in the GET array
  173.   if (isset($_SERVER['PATH_INFO'])) {
  174.     // split the request by /
  175.     $raw_data = explode('/', $_SERVER['PATH_INFO']);
  176.     // get rid of empty values
  177.     $raw_data = array_diff($raw_data, array(''));
  178.     // how many values we have
  179.     $total_data = count($raw_data);
  180.     // if total number of vaues divides by 2, then we have pairs (name-value) so we can start to do our stufsf
  181.     if (is_int($total_data / 2)) {
  182.       for($i=1;$i<=count($raw_data);$i+=2) {
  183.         $_GET[$raw_data[$i]] = $raw_data[$i+1];
  184.       }
  185.       if(isset($_SERVER['REDIRECT_QUERY_STRING'])) {
  186.         $new_qs = trim($_SERVER['REDIRECT_QUERY_STRING']);
  187.         $new_qs = str_replace('&', '/', $new_qs);
  188.         $new_qs = str_replace('=', '/', $new_qs);
  189.         $request_uri = str_replace('?'.$_SERVER['REDIRECT_QUERY_STRING'], '/'.$new_qs, $_SERVER['REQUEST_URI']);
  190.         if(isset($_SERVER['HTTP_HOST']) && $_SERVER['HTTP_HOST'] != '') {$server = $_SERVER['HTTP_HOST'];}
  191.         else if(isset($_SERVER['SERVER_NAME']) && $_SERVER['SERVER_NAME'] != '') {$server = $_SERVER['SERVER_NAME'];}
  192.         $location = 'http://' . $server . $request_uri;
  193.         header("Location: " . $location);
  194.       }
  195.     }
  196.     // if total number of values doesn't divide by 2, the request is invalid so we redirect to index
  197.     else {
  198.       header("Location: " . _APP_URL);
  199.     }
  200.   }
  201. }

  775. function pmc_filter_input($input) {
  776.   if(get_magic_quotes_gpc()) {$input = stripslashes($input);}
  777.   // get rid of unwanted characters
  778.   $unwanted = array('\\', '');
  779.   $input = str_replace($unwanted, ' ', $input);
  780.   // block tags

  787. /**
  788. * Filter user input for searching
  789. *
  790. * @return string filtered input
  791. */
  792. function pmc_filter_input_search($input) {
  793.   $input  = pmc_filter_input($input);
  794.   $common = array(' the ', ' and ', ' or ');
  795.   $input  = str_replace($common, ' ', $input);
  796.   // split by words
  797.   $words = preg_split("/[\s,]+/", $input);
  798.   // get rid of duplicates
  799.   $words = array_unique($words);
  800.   // get rid of short words 1, 2 chars
  801.   $words = preg_replace("/^([a-z]{1,2})$/", '', $words);
  802.   // remove some unwanted chars
  803.   $words = preg_replace("/^([.']{1,})$/", '', $words);
  804.   // get rid of empty words
  805.   $words = array_diff($words, array(''));
  806.   // create string
  807.   $input = implode('|', $words);
  808.   return $input;
  809. }
